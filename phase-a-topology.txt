// ---------- 名前空間 ----------
window._topoPhaseA_state = {
  clusters: null,   // Map<clusterId, {x, y, edges:[{polyIdx, vertIdx}], type}>
  edges: null,      // Array<{from:clusterId, to:clusterId, polyIdx, vertRange:[a,b]}>
  overlayLayer: null,
  lastParams: null,
};

// ---------- SpatialHash ----------
class _TopoSpatialHash {
  constructor(cellSize) {
    this.cellSize = cellSize;
    this.buckets = new Map();
  }
  _key(x, y) {
    return Math.floor(x / this.cellSize) + ',' + Math.floor(y / this.cellSize);
  }
  add(x, y, payload) {
    const k = this._key(x, y);
    if (!this.buckets.has(k)) this.buckets.set(k, []);
    this.buckets.get(k).push({ x, y, payload });
  }
  // 半径 r 以内の近傍を返す（自身も含む）
  queryRadius(x, y, r) {
    const cx = Math.floor(x / this.cellSize);
    const cy = Math.floor(y / this.cellSize);
    const range = Math.ceil(r / this.cellSize);
    const r2 = r * r;
    const hits = [];
    for (let dx = -range; dx <= range; dx++) {
      for (let dy = -range; dy <= range; dy++) {
        const k = (cx + dx) + ',' + (cy + dy);
        const bucket = this.buckets.get(k);
        if (!bucket) continue;
        for (const p of bucket) {
          const ddx = p.x - x, ddy = p.y - y;
          if (ddx * ddx + ddy * ddy <= r2) hits.push(p);
        }
      }
    }
    return hits;
  }
}

// ---------- 入力取得: Potrace 結果 → polylines ----------
function _topoPhaseA_getPolylines() {
  if (typeof _potraceLastResult === 'undefined' || !_potraceLastResult) {
    throw new Error('Potrace 結果が見つかりません。先に Potrace 実行してください。');
  }
  // _potraceLastResult.svg または .paths から頂点配列を取得
  // 実装は既存の _potracePathToVertices を利用
  const svgStr = _potraceLastResult.svg;
  if (!svgStr) throw new Error('Potrace SVG が空です。');
  const parser = new DOMParser();
  const doc = parser.parseFromString(svgStr, 'image/svg+xml');
  const pathEls = doc.querySelectorAll('path');
  const polylines = [];
  pathEls.forEach((el, idx) => {
    const d = el.getAttribute('d');
    if (!d) return;
    const verts = _potracePathToVertices(d); // 既存関数
    if (verts && verts.length >= 2) polylines.push(verts);
  });
  return polylines;
}

// ---------- クラスタリング: 頂点を近接統合 ----------
function _topoPhaseA_clusterVertices(polylines, snapDist) {
  const hash = new _TopoSpatialHash(snapDist * 2);
  const clusters = new Map(); // clusterId → {x, y, count, edges:[]}
  const vertexToCluster = []; // vertexToCluster[polyIdx][vertIdx] = clusterId
  let nextId = 0;

  for (let pi = 0; pi < polylines.length; pi++) {
    const poly = polylines[pi];
    vertexToCluster[pi] = new Array(poly.length);
    for (let vi = 0; vi < poly.length; vi++) {
      const v = poly[vi];
      const near = hash.queryRadius(v.x, v.y, snapDist);
      if (near.length === 0) {
        const cid = nextId++;
        clusters.set(cid, { x: v.x, y: v.y, count: 1, edges: [] });
        hash.add(v.x, v.y, { cid });
        vertexToCluster[pi][vi] = cid;
      } else {
        // 最も近いクラスタに参加
        let best = near[0];
        let bestD2 = (best.x - v.x) ** 2 + (best.y - v.y) ** 2;
        for (let k = 1; k < near.length; k++) {
          const d2 = (near[k].x - v.x) ** 2 + (near[k].y - v.y) ** 2;
          if (d2 < bestD2) { best = near[k]; bestD2 = d2; }
        }
        const cid = best.payload.cid;
        const c = clusters.get(cid);
        // 重心更新
        c.x = (c.x * c.count + v.x) / (c.count + 1);
        c.y = (c.y * c.count + v.y) / (c.count + 1);
        c.count += 1;
        vertexToCluster[pi][vi] = cid;
      }
    }
  }
  return { clusters, vertexToCluster };
}

// ---------- エッジ抽出: polyline をクラスタ境界で分割 ----------
function _topoPhaseA_extractEdges(polylines, vertexToCluster, clusters) {
  const edges = [];
  for (let pi = 0; pi < polylines.length; pi++) {
    const v2c = vertexToCluster[pi];
    let startVi = 0;
    let startCid = v2c[0];
    for (let vi = 1; vi < v2c.length; vi++) {
      const cid = v2c[vi];
      if (cid !== startCid && vi < v2c.length) {
        // startCid と cid の間のエッジを記録
        edges.push({
          from: startCid,
          to: cid,
          polyIdx: pi,
          vertRange: [startVi, vi],
        });
        // クラスタに接続情報を登録（一時的に polyIdx と vertIdx で保持）
        clusters.get(startCid).edges.push({ polyIdx: pi, vertIdx: startVi, dir: 'out', otherCid: cid, edgeIdx: edges.length - 1 });
        clusters.get(cid).edges.push({ polyIdx: pi, vertIdx: vi, dir: 'in', otherCid: startCid, edgeIdx: edges.length - 1 });
        startVi = vi;
        startCid = cid;
      }
    }
  }
  return edges;
}

// ---------- 角度計算ユーティリティ ----------
function _topoPhaseA_edgeAngleAt(polylines, cluster, edgeInfo) {
  // クラスタから見た、エッジの初期方向（角度 rad）
  const poly = polylines[edgeInfo.polyIdx];
  const vi = edgeInfo.vertIdx;
  // dir='out' なら vi → vi+1 方向、'in' なら vi → vi-1 方向
  let dx, dy;
  if (edgeInfo.dir === 'out') {
    if (vi + 1 >= poly.length) return null;
    dx = poly[vi + 1].x - poly[vi].x;
    dy = poly[vi + 1].y - poly[vi].y;
  } else {
    if (vi - 1 < 0) return null;
    dx = poly[vi - 1].x - poly[vi].x;
    dy = poly[vi - 1].y - poly[vi].y;
  }
  return Math.atan2(dy, dx);
}

function _topoPhaseA_normalizeAngleDeg(deg) {
  let d = deg % 360;
  if (d > 180) d -= 360;
  if (d < -180) d += 360;
  return d;
}

// ---------- 5類型判定 ----------
function _topoPhaseA_classify(cluster, polylines, params) {
  const edges = cluster.edges;
  const degree = edges.length;

  if (degree < 2) return 'endpoint'; // 端点（未分類扱い）
  if (degree > 4) return 'complex';  // 5本以上は複雑（未分類扱い）

  // 各エッジの初期方向を計算
  const angles = [];
  for (const e of edges) {
    const a = _topoPhaseA_edgeAngleAt(polylines, cluster, e);
    if (a !== null) angles.push(a * 180 / Math.PI);
  }
  if (angles.length !== degree) return 'unknown';

  if (degree === 4) {
    return 'cross'; // ② 十字
  }

  if (degree === 3) {
    // ③T字 vs ⑤3線分岐
    // 3辺の角度を比較。2本が「直線的に連続」= 相互 ±180°、残り1本が直交気味 → T字
    const tol = params.angle90Tol;
    for (let i = 0; i < 3; i++) {
      for (let j = i + 1; j < 3; j++) {
        const diff = Math.abs(_topoPhaseA_normalizeAngleDeg(angles[i] - angles[j] + 180));
        if (diff < tol) {
          // i と j が対向（直線的）→ 残りが枝
          const kth = 3 - i - j;
          const branchAng1 = Math.abs(_topoPhaseA_normalizeAngleDeg(angles[kth] - angles[i] - 90));
          const branchAng2 = Math.abs(_topoPhaseA_normalizeAngleDeg(angles[kth] - angles[i] + 90));
          if (Math.min(branchAng1, branchAng2) < tol) return 't_junction'; // ③ T字
          return 'branch_3'; // ⑤ 3線分岐
        }
      }
    }
    return 'branch_3'; // ⑤ 3線分岐
  }

  if (degree === 2) {
    // ①R付きコーナー vs ④90°折れ
    let angDiff = Math.abs(_topoPhaseA_normalizeAngleDeg(angles[0] - angles[1] + 180));
    // ここで angDiff = 0 なら直線通過（実質エッジ内、稀）、90 なら直角、それ以外は角度
    const bendAngle = 180 - angDiff; // 折れ角
    const tol = params.angle90Tol;

    // R付き判定: 近傍数頂点の累積角度変化を見る
    const rInfo = _topoPhaseA_checkRCorner(cluster, polylines, params);
    if (rInfo.isRCorner) return 'corner_r'; // ① R付きコーナー

    if (Math.abs(bendAngle - 90) < tol) return 'corner_90'; // ④ 90°折れ

    return 'corner_other'; // 直線通過や斜め折れ（未分類扱い）
  }

  return 'unknown';
}

// ---------- R付きコーナー判定 ----------
function _topoPhaseA_checkRCorner(cluster, polylines, params) {
  // クラスタ位置の近傍（半径 rDistMax）で、累積角度変化 > rCumDeg なら R付き
  const rDistMax = params.rDistMax;
  const rCumDeg = params.rCumDeg;

  for (const e of cluster.edges) {
    const poly = polylines[e.polyIdx];
    const startVi = e.vertIdx;
    let totalAng = 0;
    let dist = 0;
    let prev = null;
    const step = e.dir === 'out' ? 1 : -1;
    for (let k = 0; k < 10; k++) {
      const vi = startVi + step * k;
      if (vi < 0 || vi >= poly.length - 1) break;
      const v0 = poly[vi];
      const v1 = poly[vi + step];
      if (!v1) break;
      dist += Math.hypot(v1.x - v0.x, v1.y - v0.y);
      if (dist > rDistMax) break;
      const ang = Math.atan2(v1.y - v0.y, v1.x - v0.x) * 180 / Math.PI;
      if (prev !== null) {
        totalAng += Math.abs(_topoPhaseA_normalizeAngleDeg(ang - prev));
      }
      prev = ang;
    }
    if (totalAng > rCumDeg) {
      return { isRCorner: true, cumDeg: totalAng };
    }
  }
  return { isRCorner: false };
}

// ---------- モーダル制御 ----------
function _topoPhaseA_openModal() {
  const m = document.getElementById('topo-modal');
  if (!m) { alert('topo-modal 要素が見つかりません。HTML 追加が未反映です。'); return; }
  m.style.display = 'flex';
}
function _topoPhaseA_closeModal() {
  const m = document.getElementById('topo-modal');
  if (m) m.style.display = 'none';
}

// ---------- 実行 ----------
function _topoPhaseA_run() {
  const params = {
    snapDist: parseFloat(document.getElementById('topo-p-snap').value),
    angle90Tol: parseFloat(document.getElementById('topo-p-angle').value),
    rCumDeg: parseFloat(document.getElementById('topo-p-rcum').value),
    rDistMax: parseFloat(document.getElementById('topo-p-rdist').value),
  };
  console.log('[TopoPhaseA] params:', params);

  let polylines;
  try {
    polylines = _topoPhaseA_getPolylines();
  } catch (e) {
    alert(e.message);
    return;
  }
  console.log('[TopoPhaseA] input polylines:', polylines.length);

  const t0 = performance.now();
  const { clusters, vertexToCluster } = _topoPhaseA_clusterVertices(polylines, params.snapDist);
  const t1 = performance.now();
  const edges = _topoPhaseA_extractEdges(polylines, vertexToCluster, clusters);
  const t2 = performance.now();

  // 分類
  const typeCounts = {};
  for (const [cid, c] of clusters) {
    c.type = _topoPhaseA_classify(c, polylines, params);
    typeCounts[c.type] = (typeCounts[c.type] || 0) + 1;
  }
  const t3 = performance.now();

  _topoPhaseA_state.clusters = clusters;
  _topoPhaseA_state.edges = edges;
  _topoPhaseA_state.lastParams = params;

  console.log('[TopoPhaseA] clusters:', clusters.size, 'edges:', edges.length);
  console.log('[TopoPhaseA] type counts:', typeCounts);
  console.log('[TopoPhaseA] timing: cluster=' + (t1 - t0).toFixed(1) + 'ms, edges=' + (t2 - t1).toFixed(1) + 'ms, classify=' + (t3 - t2).toFixed(1) + 'ms');

  // 統計表示
  const statsEl = document.getElementById('topo-stats');
  const totalVerts = polylines.reduce((s, p) => s + p.length, 0);
  statsEl.style.display = 'block';
  statsEl.innerHTML = `
    <div><b>入力</b>: ${polylines.length} polylines, ${totalVerts} 頂点</div>
    <div><b>クラスタ</b>: ${clusters.size} (統合率 ${(100 * (1 - clusters.size / totalVerts)).toFixed(1)}%)</div>
    <div><b>エッジ</b>: ${edges.length}</div>
    <div><b>分類内訳</b>:</div>
    <div style="margin-left:12px;">
      ${Object.entries(typeCounts).map(([k, v]) => `${_topoPhaseA_typeLabel(k)}: ${v}`).join(' / ')}
    </div>
    <div><b>処理時間</b>: cluster ${(t1 - t0).toFixed(0)}ms / edges ${(t2 - t1).toFixed(0)}ms / classify ${(t3 - t2).toFixed(0)}ms</div>
  `;

  // 可視化
  _topoPhaseA_drawOverlay();
}

function _topoPhaseA_typeLabel(t) {
  return ({
    cross: '十字',
    t_junction: 'T字',
    corner_90: '90°折れ',
    corner_r: 'R付き',
    branch_3: '3線分岐',
    corner_other: '斜め角',
    endpoint: '端点',
    complex: '5本以上',
    unknown: '未分類',
  })[t] || t;
}

function _topoPhaseA_typeColor(t) {
  return ({
    cross: '#e53935',
    t_junction: '#fb8c00',
    corner_90: '#43a047',
    corner_r: '#1e88e5',
    branch_3: '#8e24aa',
    corner_other: '#795548',
    endpoint: '#607d8b',
    complex: '#000000',
    unknown: '#9e9e9e',
  })[t] || '#9e9e9e';
}

// ---------- 可視化: CAD キャンバスにマーカー描画 ----------
function _topoPhaseA_drawOverlay() {
  _topoPhaseA_clearOverlay();
  const st = _topoPhaseA_state;
  if (!st.clusters) return;

  // 既存 CAD 座標系→スクリーン変換関数を利用（前提: worldToScreen(x,y) がある）
  // 存在しない場合は既存のズーム/パン変数から自前で計算
  const overlay = document.createElement('canvas');
  overlay.id = '_topoPhaseA_overlay';
  const mainCanvas = document.getElementById('cad-canvas') || document.querySelector('canvas');
  if (!mainCanvas) { console.warn('[TopoPhaseA] canvas not found for overlay'); return; }
  overlay.width = mainCanvas.width;
  overlay.height = mainCanvas.height;
  overlay.style.cssText = 'position:absolute; left:' + mainCanvas.offsetLeft + 'px; top:' + mainCanvas.offsetTop + 'px; pointer-events:none; z-index:100;';
  mainCanvas.parentNode.appendChild(overlay);
  st.overlayLayer = overlay;

  const ctx = overlay.getContext('2d');
  ctx.clearRect(0, 0, overlay.width, overlay.height);

  // 座標変換: 既存の worldToScreen があればそれを、なければグローバル _cadTransform を参照
  const w2s = (typeof worldToScreen === 'function') ? worldToScreen
             : (typeof _cadTransform === 'object' && _cadTransform)
             ? (x, y) => ({ x: (x - _cadTransform.offsetX) * _cadTransform.scale, y: (y - _cadTransform.offsetY) * _cadTransform.scale })
             : null;
  if (!w2s) {
    console.warn('[TopoPhaseA] 座標変換関数が見つかりません。worldToScreen または _cadTransform をご確認ください。');
    return;
  }

  for (const [cid, c] of st.clusters) {
    const s = w2s(c.x, c.y);
    ctx.beginPath();
    ctx.arc(s.x, s.y, 6, 0, Math.PI * 2);
    ctx.fillStyle = _topoPhaseA_typeColor(c.type);
    ctx.fill();
    ctx.lineWidth = 1;
    ctx.strokeStyle = '#000';
    ctx.stroke();
  }
}

function _topoPhaseA_clearOverlay() {
  const el = document.getElementById('_topoPhaseA_overlay');
  if (el) el.remove();
  _topoPhaseA_state.overlayLayer = null;
}
